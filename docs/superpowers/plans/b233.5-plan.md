// AdmitCarrier 对一次无小队的载体直派只占载体物理位。
// Binding.Squad 为空；HomeDir 是绑定当时 IdentityOf 快照。
func (s *Service) AdmitCarrier(name string) (Binding, error) {
	name = strings.TrimSpace(name)
	statusLog().Info("scheduling.admit_carrier.start", "carrier", name)
	c, err := s.Carrier(name)
	if err != nil {
		statusLog().Error("scheduling.admit_carrier.error", "carrier", name, "error_kind", "lookup", "cause", err)
		return Binding{}, err
	}
	if c.Status != StatusOnline {
		statusLog().Warn("scheduling.admit_carrier.error", "carrier", name, "status", c.Status, "error_kind", "no_healthy")
		return Binding{}, ErrNoHealthy
	}
	_, carrierKey := OccupancyKeys("", c.Name)
	for attempt := 0; attempt < maxCASAttempts; attempt++ {
		running, expect, err := s.readCount(carrierKey)
		if err != nil {
			return Binding{}, err
		}
		if c.MaxConcurrency > 0 && running >= c.MaxConcurrency {
			statusLog().Warn("scheduling.admit_carrier.full", "carrier", c.Name, "carrier_cap", c.MaxConcurrency)
			return Binding{}, ErrNoSlot
		}
		if err := s.casCount(carrierKey, expect, running+1); err != nil {
			if errors.Is(err, schedclient.ErrCASConflict) {
				continue
			}
			return Binding{}, err
		}
		id := IdentityOf(c)
		b := Binding{Squad: "", Carrier: c.Name, Target: id.Machine, Executor: id.CLI, Model: c.Model, HomeDir: id.HomeDir}
		statusLog().Info("scheduling.admit_carrier.success", "carrier", c.Name, "target", b.Target, "cli", b.Executor)
		return b, nil
	}
	return Binding{}, fmt.Errorf("%w: %s", ErrRetryExhausted, carrierKey)
}

func (s *Service) DefaultCarrier() (string, error) {
	rec, err := s.repo.Get(KindDefaultCarrier, DefaultCarrierID)
	if err != nil {
		if errors.Is(err, schedclient.ErrNotFound) {
			return "", ErrNoDefault
		}
		return "", err
	}

	if err != nil {
		return "", err
	}
	if _, err := s.Carrier(name); err != nil {
		return "", fmt.Errorf("%w: 默认载体 %s 未登记", ErrNoDefault, name)
	}
	return name, nil
}

// SetDefaultCarrier 写入默认载体 singleton。空名或非载体名拒绝。
func (s *Service) SetDefaultCarrier(name string) error {
	name = strings.TrimSpace(name)
	body, err := EncodeDefaultCarrier(name)
	if err != nil {
		return err
	}
	if _, err := s.Carrier(name); err != nil {
		if errors.Is(err, ErrNotFound) {
			return fmt.Errorf("%w: 默认载体 %s 未登记为载体", ErrNoDefault, name)
		}
		return err
	}
	expect := 0
	if rec, getErr := s.repo.Get(KindDefaultCarrier, DefaultCarrierID); getErr == nil {
		expect = rec.Version
	} else if !errors.Is(getErr, schedclient.ErrNotFound) {
		return getErr

	if err != nil {
		statusLog().Error("写入默认载体失败", "name", name, "expect", expect, "cause", err)
		return err
	}
	statusLog().Info("写入默认载体成功", "name", name, "expect", expect)

// receiver_bind.go —— 裸派发与卡节点共用的接收者准备链（B233.5）。
//
// 职责：ResolveLookup → Admit/AdmitCarrier → BindPhysical → EffectiveModel。
// 边界：不启动任务、不写 Task、不读卡列。Manager 不得调用本文件。
package agentd

func (s *Server) prepareReceiverBinding(requested string, overlay scheduling.PhysicalOverlay, requestModel string) (scheduling.Binding, scheduling.ResolvedReceiver, error) {
	if s.scheduling == nil {
		return scheduling.Binding{}, scheduling.ResolvedReceiver{}, fmt.Errorf("编制域服务未装配（SetupAutomation 未执行）")
	}
	defaultName, defErr := s.scheduling.DefaultCarrier()
	if defErr != nil && strings.TrimSpace(requested) == "" {
		s.log.Warn("派发无有效默认载体", "requested", requested, "cause", defErr)
		return scheduling.Binding{}, scheduling.ResolvedReceiver{}, defErr
	}
	if defErr != nil {
		defaultName = ""
	}
	var lookupErr error
	lookup := func(name string) (bool, bool) {
		_, cErr := s.scheduling.Carrier(name)
		_, qErr := s.scheduling.Squad(name)
		if cErr != nil && !errors.Is(cErr, scheduling.ErrNotFound) {
			lookupErr = cErr
		}
		if qErr != nil && !errors.Is(qErr, scheduling.ErrNotFound) && lookupErr == nil {
			lookupErr = qErr
		}
		return cErr == nil, qErr == nil
	}
	resolved, err := scheduling.ResolveLookup(requested, defaultName, lookup)
	if lookupErr != nil {
		return scheduling.Binding{}, scheduling.ResolvedReceiver{}, lookupErr
	}
	if err != nil {
		s.log.Warn("接收者解析失败", "requested", requested, "default", defaultName, "cause", err)
		return scheduling.Binding{}, scheduling.ResolvedReceiver{}, err
	}
	s.log.Info("接收者已解析", "requested", requested, "name", resolved.Name,
		"kind", string(resolved.Kind), "via_default", resolved.ViaDefault)
	var binding scheduling.Binding
	switch resolved.Kind {
	case scheduling.ReceiverCarrier:
		binding, err = s.scheduling.AdmitCarrier(resolved.Name)
	case scheduling.ReceiverSquad:
		binding, err = s.scheduling.Admit(scheduling.IgnitionRequest{Squad: resolved.Name, Model: requestModel, Actor: "dispatch"})
	default:
		err = fmt.Errorf("%w: 未知接收者类型 %s", scheduling.ErrInvalid, resolved.Kind)
	}
	if err != nil {
		s.log.Error("准入失败", "name", resolved.Name, "kind", string(resolved.Kind), "cause", err)
		return scheduling.Binding{}, resolved, err
	}
	id := scheduling.PhysicalIdentity{Machine: binding.Target, CLI: binding.Executor, HomeDir: binding.HomeDir}
	bound, err := scheduling.BindPhysical(id, overlay)
	if err != nil {
		s.log.Warn("物理覆盖被拒", "carrier", binding.Carrier, "cause", err)
		if relErr := s.scheduling.Release(binding.Squad, binding.Carrier); relErr != nil {
			s.log.Error("覆盖拒绝后释放占用失败", "carrier", binding.Carrier, "cause", relErr)
		}
		return scheduling.Binding{}, resolved, err
	}
	binding.Target = bound.Machine
	binding.Executor = bound.CLI
	binding.HomeDir = bound.HomeDir
	binding.Model = scheduling.EffectiveModel(binding.Model, requestModel)
	s.log.Info("接收者已绑定", "carrier", binding.Carrier, "squad", binding.Squad,
		"machine", binding.Target, "cli", binding.Executor, "home_dir", binding.HomeDir,
		"model", binding.Model, "via_default", resolved.ViaDefault)
	return binding, resolved, nil
}
