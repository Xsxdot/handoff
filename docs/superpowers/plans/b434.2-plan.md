//   - cardSeatDesktopSetCmd（T3）：以卡上当前 driver_session 为 CAS 期望值补写记录。
//
// 边界：只经 ledger.Store 的公开写面写账本；不读不写 seat_bearings；不调 agentd；
// 不改席位真源 cards.driver_session。契约见 b434-contract.md §3.3、§3.10。

// desktopSeatCLI 是唯一能挂桌面端记录 / 登记拉起器的席位物种：Claude 桌面端会话只跑
// claude（契约条 60、141）。
const desktopSeatCLI = "claude"

// desktopSeatInput 是 --desktop-session / --machine / --via 的原值与 --machine 是否显式给出。
type desktopSeatInput struct {
	Session        string
	Machine        string
	MachineChanged bool
	Via            string
}

// validSeatDesktopViaFlag 报告 --via 是否在 proto.SeatDesktopVia* 词表内（大小写敏感）。
// 账本写闸也判同一词表；CLI 先判是为了不让词表外的值留到两步提交的第二步才被拒。
func validSeatDesktopViaFlag(via string) bool {
	switch via {
	case proto.SeatDesktopViaPage, proto.SeatDesktopViaLauncher, proto.SeatDesktopViaManual:
		return true
	}
	return false
}

// prepareSeatDesktop 校验桌面端记录入参并补缺省值。
//
// 参数：identity 是将要（bind）或已经（seat desktop set）坐在卡上的席位身份
// cli:<cli>#<id>；in 是 flag 原值。
// 返回：desk 可直接交给 ledger.Store.SetSeatDesktop（BoundAt 留零值，账本取时钟）；
// hint 非空时调用方原样打到 stderr（机器名提示，不影响成败）；err 非 nil 时调用方
// 必须在任何账本写入之前返回。
// 注意：--via 由 flag 缺省值给出 manual（契约条 47）；显式空串按词表外拒绝。
func prepareSeatDesktop(identity string, in desktopSeatInput) (ledger.SeatDesktop, string, error) {
	cli, _, err := proto.ParseSeatIdentity(identity)
	if err != nil {
		return ledger.SeatDesktop{}, "", fmt.Errorf("解析席位身份: %w", err)
	}
	if cli != desktopSeatCLI {
		return ledger.SeatDesktop{}, "", fmt.Errorf("--desktop-session 只能用于 claude 会话的席位，当前席位物种是 %q", cli)
	}
	if err := proto.ValidateDesktopSessionID(in.Session); err != nil {
		return ledger.SeatDesktop{}, "", fmt.Errorf("--desktop-session 非法: %w", err)
	}
	if !validSeatDesktopViaFlag(in.Via) {
		return ledger.SeatDesktop{}, "", fmt.Errorf("--via 只接受 page、launcher、manual，得到 %q", in.Via)
	}
	machine, hint, err := resolveDesktopMachine(in.Machine, in.MachineChanged)
	if err != nil {
		return ledger.SeatDesktop{}, "", err
	}
	return ledger.SeatDesktop{Session: in.Session, Machine: machine, Via: in.Via}, hint, nil
}

// resolveDesktopMachine 决定桌面端记录 / 拉起器登记的机器名。
//
// 参数：flagValue 是 --machine 原值；changed 表示命令行显式给了 --machine（显式空串照收，
// 账本里空串 = 本机）。
// 返回：machine 为显式值或 os.Hostname()；hint 非空表示缺省取了 hostname，而本机配置
// 有 targets 却没有同名项——其它 agentd 异机转交按 target 名寻址（契约 §11.2），名字
// 对不上转交会失败；err 仅在未显式给值且取不到 hostname 时返回。
// 注意：只提示不报错；单机部署（无 targets）不提示。
func resolveDesktopMachine(flagValue string, changed bool) (string, string, error) {
	if changed {
		return flagValue, "", nil
	}
	host, err := os.Hostname()
	if err != nil {
		return "", "", fmt.Errorf("取本机 hostname 作 --machine 缺省值失败（可显式给 --machine）: %w", err)
	}
	cfg := loadCLIConfig()
	if len(cfg.Targets) > 0 {
		if _, ok := cfg.Targets[host]; !ok {
			return host, fmt.Sprintf("提示：--machine 缺省取本机 hostname %q；其它机器的 agentd 按 target 名把唤醒转交到这里，若它们配置里本机的 target 名不是 %q，请用 --machine <那个名字> 重新登记", host, host), nil
		}
	}
	return host, "", nil
}

func (c *Client) DesktopLauncherOpen(ctx context.Context, cardID string) (*proto.DesktopLauncherOpenResp, error) {
	c.log().Info("请求拉起器代开桌面端协调者", "card", cardID)
	resp, err := c.do(ctx, http.MethodPost,
		"/api/cards/"+url.PathEscape(cardID)+"/coordinator/desktop-launch", struct{}{})
	if err != nil {
		c.log().Warn("拉起器代开请求未拿到响应", "card", cardID, "cause", err)
		return nil, err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return nil, c.httpError("desktop launcher open", resp)
	}
	var out proto.DesktopLauncherOpenResp
	if err := decodeWire(resp, &out); err != nil {
		c.log().Warn("拉起器代开回执解码失败", "card", cardID, "cause", err)
		return nil, err
	}
	c.log().Info("拉起器代开已受理", "card", cardID,
		"launcher", out.Launcher, "machine", out.Machine, "handled_by", out.HandledBy)
	return &out, nil
}

// ServerErrorText 取出 agentd 非 2xx 响应里给人看的错误原文。
//
// 参数：err 任意错误（通常来自 Client 方法）。
// 返回：ok=false 表示 err 链上没有 agentd 的 HTTP 状态错误（连接失败、解码失败等），
// 调用方按原错误处理；ok=true 时 msg 依次取：响应体是 JSON 对象且 `error` 为非空字符串
// → 该字符串；否则 → 去首尾空白后的原始响应体；原始体也为空 → 「agentd 返回状态码 N」。
// 注意：响应体已被 httpError 截在 4096 字节内；本函数只投影文本，不改错误本身。
// 为什么需要它：CLI 要把服务端的可行动文案（如「无拉起器，请在桌面端开一场并登记」）
// 原样交给人，不带 {"error":…} JSON 外壳（契约条 258–259）；httpStatusError 不导出，
// cmd 层拿不到 body。
func ServerErrorText(err error) (string, bool) {
	var he *httpStatusError
	if !errors.As(err, &he) {
		return "", false
	}
	var body struct {
		Error *string `json:"error"`
	}
	if json.Unmarshal([]byte(he.body), &body) == nil && body.Error != nil && strings.TrimSpace(*body.Error) != "" {
		return *body.Error, true
	}
	if he.body != "" {
		return he.body, true
	}
	return fmt.Sprintf("agentd 返回状态码 %d", he.code), true
}

// probeLauncherMachine 判一台拉起器机器够不够得着。
//
// 参数：cfg 为本机 CLI 配置；host 为本机 hostname（取不到为空串）；machine 为登记值。
// 返回：reach ∈ {可达, 不可达（原因）, 不可探}；warn 非空表示机器名不利于异机寻址。
// 判定顺序：本机配置 targets 里有同名项 → 经该 target 探；空串或等于 hostname → 探本机
// agentd（LocalEndpoint，不受 --target 影响）；其它名字 → 不可探 + 告警。

// 契约方向；且 local / 本机 这类别名在异机发起端会被解释成发起机自己——对拉起器登记而言
// 恰是该告警的错配，空串同理（契约 §11.2）。
func probeLauncherMachine(ctx context.Context, cfg *config.Config, host, machine string) (string, string) {
	ctx, cancel := context.WithTimeout(ctx, launcherProbeTimeout)
	defer cancel()
	if _, ok := cfg.Targets[machine]; ok && machine != "" {
		c, done, err := newTargetClientNamed(machine)
		if err != nil {
			return fmt.Sprintf("不可达（%v）", err), ""
		}
		defer done()
		return agentdReach(ctx, c), ""
	}
	if machine == "" || (host != "" && machine == host) {
		addr, token, err := LocalEndpoint()
		if err != nil {
			return fmt.Sprintf("不可达（%v）", err), ""
		}
		reach := agentdReach(ctx, client.New(addr, token))
		if machine == "" {
			return reach, "机器名为空：异机发起的代开会把它当成发起机自己，请用 --machine <本机 target 名> 重新登记"
		}
		return reach, ""
	}
	return "不可探", fmt.Sprintf("机器名 %q 既不是本机 hostname 也不在本机配置 targets 里；异机转交按 target 名寻址，请确认它与各 agentd 的 targets 一致", machine)

// skillSection 返回以 headingPrefix 开头的标题行起、到下一个 #/##/### 标题之前的正文；
// 围栏代码块内的行不当标题（片段里可能有 # 开头的行）。找不到即 Fatal。
func skillSection(t *testing.T, text, headingPrefix string) string {
	t.Helper()
	lines := strings.Split(text, "\n")
	start := -1
	for i, ln := range lines {
		if strings.HasPrefix(ln, headingPrefix) {
			start = i
			break
		}
	}
	if start < 0 {
		t.Fatalf("SKILL.md 缺少以 %q 开头的标题", headingPrefix)
	}
	end := len(lines)
	inFence := false
	for i := start + 1; i < len(lines); i++ {
		ln := lines[i]
		if strings.HasPrefix(strings.TrimSpace(ln), "```") {
			inFence = !inFence
			continue
		}
		if !inFence && (strings.HasPrefix(ln, "# ") || strings.HasPrefix(ln, "## ") || strings.HasPrefix(ln, "### ")) {
			end = i
			break
		}
	}
	return strings.Join(lines[start:end], "\n")
}

// fencedBlockAfter 返回 section 里 marker 之后第一个围栏代码块的内容（不含开闭栏行）。
func fencedBlockAfter(t *testing.T, section, marker string) string {
	t.Helper()
	i := strings.Index(section, marker)
	if i < 0 {
		t.Fatalf("节内缺少标记 %q", marker)
	}
	var body []string
	in := false
	for _, ln := range strings.Split(section[i:], "\n") {
		if strings.HasPrefix(strings.TrimSpace(ln), "```") {
			if in {
				return strings.Join(body, "\n")
			}
			in = true
			continue
		}
		if in {
			body = append(body, ln)
		}
	}
	t.Fatalf("标记 %q 之后没有闭合的围栏代码块", marker)
	return ""
}

## 11. 需协调者知悉或裁决的点（可能越契约面）

1. **新增导出 `client.ServerErrorText(err error) (string, bool)`**（契约 §3.10 只列 `DesktopLauncherOpen`）。理由：`httpStatusError` 未导出，cmd 层拿不到 `error` 字段，条 258「不含 `{"error"` 外壳」无从实现；替代方案是让 `DesktopLauncherOpen` 返回自定义错误类型（同样新增导出面）。属 `d_cli→d_transport` 既有方向内的新入口，无新方向。**请认可或改判**。
2. **真实进程 stderr 仍含 slog 行里的 JSON 体**：`client.httpError` 对 4xx 打 `WARN agentd 请求被拒 … body={"error":…}`，`card` 族 CLI 未调 `logx.Setup`，slog 默认写进程 stderr。条 258 的测试锁的是 cobra `ErrOrStderr`（契约「断言方式：cmd 包 Go 测试」），满足字面；但真机上人看到的 stderr 会多这一行。本卡**不改** client 日志级别与 CLI 日志初始化（会波及全部 CLI 命令，越界）。建议：作为已知面记账，或另开卡统一 CLI 日志落点。
3. **`card seat desktop set` 也执法「席位 cli 段 = claude」**（契约 §3.10 只对 bind 明写）——同一规则两入口共享一道门（缺陷族·门禁绕过）；且**不要求调用方是席位本人**（人可在终端代填；契约 expect 只取卡上席位）。
4. **`card bind` 的桌面端入参在 `BindSeat` 之前全部校验**：条 60 只要求「不写桌面端记录」，本卡同时不坐下（更严）；条 61 本就要求不坐下。
5. **`card seat desktop set --via` 接受 page|launcher|manual 全词表**（契约写 `[--via manual]`，本稿解读为缺省值展示）。
6. **`handoff status --json` 不加拉起器键**（契约「格式归 plan」；保持既有 JSON 线格式不变）。
7. **status 段本机判定只认空串与 `os.Hostname()`**，`local`/`本机` 别名按「不可探 + 告警」显示（因 `d_cli` 不得 import `scheduling.IsLocalMachine`，且别名对异机寻址本就错配）。

9. **派单提到的 `handleDesktopLauncherOpen` 空壳在工作树里不存在**；G3 尚未注册 POST 路由。本卡按契约 §3.9 HTTP 形状 Consume，G3 落地前真机 `card coordinate --desktop` 会得到 404 原文。G3 plan 需保证非 2xx 体为 `{"error": "<原文>"}`（`writeJSON` 惯例）且 200 体三键恒出。
