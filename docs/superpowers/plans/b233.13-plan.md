## 1. 冻结边界、方向与一处必须记录的文本冲突

### 1.1 依赖方向

- `internal/orchestration` **import** `internal/agentd`（D1）；`internal/agentd` 生产文件**不得** import `internal/orchestration`。唯一同时 import 二者的是 `cmd/agentd.go`（组装点）。
- 接口归使用方：`OrchestrationClient` 定义在 gateway（`internal/agentd/orchestration_client.go`），实现是 `internal/orchestration.Manager`。
- `Server` 生产字段改为接口后，handler 对编排的全部调用都经接口；`*Manager` 只在 `cmd` `new`。

### 1.2 文本冲突（必须让 review 看见）

- 冻结物 contract §5.1 #2 / §2.2 写「`internal/orchestration` 不 import `internal/agentd`」；§13 修订记录（D1）写「`internal/orchestration` import gateway（仅取共享 DTO/助手）」。两者字面矛盾。
- 本计划按**后写的、且协调者两次裁决的 D1** 执行（orchestration import gateway）。原因：接口在 gateway 且方法参数类型是 gateway 本地 DTO（§2.1 冻结形状），实现要满足该接口就必须引用 `agentd.*`，否则接口参数类型与被实现类型不一致，Go 编译不过。
- 处置：plan 阶段记录冲突，不自行改冻结物；implement 阶段在台账显式记「按 D1，§5.1 #2/§2.2 第二分句被 §13 覆盖」。若 review 认为需回改 contract，交协调者。

### 1.3 反方向预算风险（未验证，迁移后判定）

- `codegraph/target.json` 有既有方向 `d_orchestration → d_gateway` `legacyBudget: 2`（当前实测 `--view cards-B233.13-charter` 显示 `d_orchestration->d_gateway: 1`）。
- D1 让编排包大量引用 gateway（`Hub`、`DispatchReq`/`RegisterProjectReq`/`RecoverReport`、`TerminalOutcome`、保留在 gateway 的 payload/助手）。这些引用在迁移后是否记为 `d_orchestration→d_gateway` 调用边、是否越过 2，**本节点无法在迁移前实测**。
- 判据：Task 6 跑 `codegraph check`，`fails:[]` 为放过；若该方向超预算导致 `fails` 非空，按 §4 Task 6 的规则处理（调该既有方向的 `legacyBudget` 到实测值并写 `legacyBudgetNote`，不改方向、不改 assembly），并在台账贴原始 check 输出。若工具要求的是 contract 层修订而非 target 编辑，停回协调者。

// transitClaim 执行一次状态迁移并返回「本次是否赢得了进入 to 的 CAS」。
//
// B233.13：终态占用释放的凭证。cur.State == to 是幂等成功（别人已迁到目标态），
// 返回 claimed=false；UpdateTaskState 内部 WHERE state=旧值 的 CAS 才是唯一赢家判据，
// 输家拿 store.ErrBadTransit。终态收口（工单作废 + 清扫）只在真正发生迁移时执行。
func (m *Manager) transitClaim(taskID string, to proto.TaskState, reason string) (bool, error) {
	cur, err := m.st.GetTask(taskID)
	if err != nil {
		return false, err
	}
	if cur.State == to {
		return false, nil
	}
	if err := m.st.UpdateTaskState(taskID, to); err != nil {
		return false, err
	}
	m.log.Info("任务状态迁移", "task", taskID, "from", cur.State, "to", to, "reason", reason)
	if to.IsTerminal() {
		voidTicketsWithAudit(m.st, taskID, reason, m.log)
		if m.consumeSweepOwned(taskID) {
			m.log.Debug("终态清扫已由 Done/Stop 就地完成，跳过", "task", taskID, "to", to)
		} else {
			m.sweep(taskID)
		}
	}
	return true, nil
}

func (m *Manager) Done(ctx context.Context, taskID, note string) (outcome TerminalOutcome, err error) {
	m.noteSweepOwned(taskID)
	m.log.Info("done 进入", "task", taskID)
	defer func() {
		if err != nil {
			m.log.Error("done 失败", "task", taskID, "cause", err)
		} else {
			m.log.Info("done 完成", "task", taskID, "claimed", outcome.Claimed)
		}
	}()

	cur, err := m.st.GetTask(taskID)
	if err != nil {
		return TerminalOutcome{}, err
	}
	// 契约 §5.2 #9：已 completed 的重复 Done 是幂等成功（Claimed=false, err==nil）。
	// 只有其它非 waiting_review 状态才是真错误（§5.2 #7）。
	if cur.State == proto.TaskStateCompleted {
		m.log.Info("done 已幂等完成", "task", taskID, "state", cur.State)
		return TerminalOutcome{Claimed: false}, nil
	}
	if cur.State != proto.TaskStateWaitingReview {
		m.log.Warn("done 状态不允许", "task", taskID, "state", cur.State)
		return TerminalOutcome{}, fmt.Errorf("任务 %s 状态 %s 不允许归档（需 waiting_review）: %w", taskID, cur.State, store.ErrBadTransit)
	}
	if note != "" {
		if err := m.st.SetTaskField(taskID, "done_note", note); err != nil {
			m.log.Error("写入归档说明失败", "task", taskID, "note_bytes", len(note), "cause", err)
			return TerminalOutcome{}, fmt.Errorf("写入归档说明: %w", err)
		}
		m.log.Info("归档说明已落库", "task", taskID, "note_bytes", len(note))
	}
	claimed, err := m.transitClaim(taskID, proto.TaskStateCompleted, "done")
	if err != nil {
		return TerminalOutcome{}, err
	}
	if !claimed {
		// 并发 loser：目标态已由赢家迁移完成并承担收尾；本条按幂等成功返回，
		// 不得重复发布归档事件/清理/释放占用（契约 §5.2 #9）。
		m.log.Info("done 幂等成功（并发中他人已完成迁移）", "task", taskID)
		return TerminalOutcome{Claimed: false}, nil
	}
	// 归档事件：必须在 hub.CloseTask 之前发布，否则订阅者（wait --follow）一条
	// 都收不到——而事件仍会入库，症状是等待方永远等不到归档，极难归因（B68 §4.2）。
	// 失败只打日志不阻塞：状态已迁移完，返回错误也回不去。
	if evt, aerr := m.st.AppendEvent(taskID, proto.EventTypeArchived,
		proto.ArchivedPayload{Note: note}); aerr != nil {
		m.log.Error("追加归档事件失败", "task", taskID, "cause", aerr)
	} else {
		m.hub.Publish(evt)
		m.log.Info("归档事件已发布", "task", taskID, "seq", evt.Seq, "has_note", note != "")
	}
	// 任务归档：清理审批链运行时状态，防内存 map 随归档任务无界增长（P2-5）
	m.clearApproverState(taskID)

	// 仅 Error 日志不影响归档（任务已完成，executor 残留交人工兜底）
	ad, err := m.adapterFor(taskID)
	if err != nil {
		m.log.Error("解析任务执行者失败", "task", taskID, "cause", err)
	} else {
		m.stopExecutor(taskID, ad)
	}
	// 终态清扫（B103）：在 worktree 清理之前，否则活进程把 cwd 钉在工作树里会让删除失败

		Managed:  cur.WorktreeManaged,
		Manual:   false,
		Terminal: true,
		State:    string(proto.TaskStateCompleted),
		Trigger:  workspace.TriggerDone,
	})
	m.log.Info("done 回收判据", "task", taskID, "decision", dec,
		"managed", cur.WorktreeManaged, "workdir", cur.WorkDir)
	if dec == workspace.RetainRecycle {
		if err := m.RequireWorkspace(); err != nil {
			m.log.Error("done 需回收但工作区能力未注入", "task", taskID, "cause", err)
		} else if werr := m.ws.RecycleManaged(ctx, cur.RepoPath, cur.WorkDir); werr != nil {
			m.log.Error("清理 managed worktree 失败", "task", taskID, "workdir", cur.WorkDir, "cause", werr)
			if evt, aerr := m.st.AppendEvent(taskID, proto.EventTypeProgress, progressPayload{
				Text: worktreeCleanupHint(taskID, werr),
			}); aerr != nil {

	}
